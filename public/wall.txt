import React, { useEffect, useRef } from "react";
import { PortfolioProvider, usePortfolio } from "../context/PortfolioContext.jsx";
import { FrameHeader } from "../components/FrameHeader.jsx";
import { EditorShell } from "../components/EditorShell.jsx";

import { identity } from "../config/portfolio.js";
import { mountEditor } from "../engines/editorEngine.js";
function EditorWorkspace() {
  const { data, drawerOpen, setDrawerOpen, wormOn, setWormOn } = usePortfolio();
  const stageRef = useRef(null);
  const controllerRef = useRef(null);
  useEffect(() => {
    if (!data || !stageRef.current) return undefined;
    const ctrl = mountEditor({
      root: stageRef.current,
      data,
      identity,
      initialWormOn: wormOn,
      onDrawerChange: (open) => setDrawerOpen(open),
    });
    controllerRef.current = ctrl;
    return () => {
      ctrl.destroy();
      controllerRef.current = null;
    };
  }, [data]);
  const handleToggleDrawer = () => {
    setDrawerOpen((prev) => {
      const next = !prev;
      controllerRef.current?.syncDrawer(next);
      return next;
    });
  };
  const handleToggleWorm = () => {
    const next = !wormOn;
    setWormOn(next);
    controllerRef.current?.setWorm(next);
  };
  return (
    <>
      <main className="stage" ref={stageRef}>
        <FrameHeader brandTip="back to the intro" />
        <h1 className="sr-only">{identity.domain} — experience, education, achievements, tools and themes.</h1>
        <EditorShell drawerOpen={drawerOpen} onToggleDrawer={handleToggleDrawer} />
        <StatusBar wormOn={wormOn} onToggleWorm={handleToggleWorm} />
        <div className="desk" aria-hidden="true" />
      </main>
      <canvas className="overlay" aria-hidden="true" />
    </>
  );
}
export function EditorPage() {
  return (
    <PortfolioProvider>
      <EditorWorkspace />
    </PortfolioProvider>
  );
}
import React, { createContext, useContext, useEffect, useState, useCallback } from "react";
import { wormStorageKey } from "../config/portfolio.js";
const PortfolioContext = createContext(null);
const loadJson = (url) =>
  fetch(url).then((response) => {
    if (!response.ok) throw new Error(`${url}: ${response.status}`);
    return response.json();
  });
export function PortfolioProvider({ children }) {
  const [data, setData] = useState(null);
  const [error, setError] = useState(null);
  const [drawerOpen, setDrawerOpen] = useState(false);
  const [wormOn, setWormOn] = useState(() => {
    try {
      return localStorage.getItem(wormStorageKey) === "on";
    } catch {
      return false;
    }
  });
  useEffect(() => {
    const controller = new AbortController();
    let active = true;
    Promise.all([
      loadJson("/site/content.json"),
      loadJson("/themes.json"),
      loadJson("/site/stack.json").catch(() => null),
    ])
      .then(([content, themeData, stack]) => {
        if (!active) return;
        setData({ content, themeData, stack });
      })
      .catch((err) => {
        if (!active || controller.signal.aborted) return;
        setError(err);
      });
    return () => {
      active = false;
      controller.abort();
    };
  }, []);
  const persistWorm = useCallback((next) => {
    setWormOn(next);
    try {
      if (next) localStorage.setItem(wormStorageKey, "on");
      else localStorage.removeItem(wormStorageKey);
    } catch {}
  }, []);
  return (
    <PortfolioContext.Provider
      value={{
        data,
        error,
        drawerOpen,
        setDrawerOpen,
        wormOn,
        setWormOn: persistWorm,
      }}
    >
      {children}
    </PortfolioContext.Provider>
  );
}
export function usePortfolio() {
  const ctx = useContext(PortfolioContext);
  if (!ctx) throw new Error("usePortfolio must be used inside PortfolioProvider");
  return ctx;
